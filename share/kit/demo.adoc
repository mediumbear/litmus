func value(for key: Key) -> String? {
    if let cached = cache[key] { return cached }
    switch store.read(key) {
    case .value(let v): cache[key] = v; return v
    case .missing: cache[key] = .some(nil); return nil
    case .locked: return nil
    }
}

== Sandbox

++++
<div id="demo-sandbox"></div>
<script src="assets/kit/sandbox.js"></script>
<script>
LitmusKit.sandbox('#demo-sandbox', {
  title: 'Upgrades and reinstalls',
  state: { defaults: ['token'], keychain: [], installed: true },
  boxes: [
    { id: 'defaults', label: 'UserDefaults' },
    { id: 'keychain', label: 'Keychain' },
  ],
  actions: [
    { label: 'Launch', when: s => s.installed, run: s => { if (s.defaults.length) { s.keychain = s.defaults; s.defaults = []; return 'The token moves to the keychain.'; } return 'Nothing to move.'; } },
    { label: 'Delete the app', when: s => s.installed, run: s => { s.defaults = []; s.installed = false; return 'UserDefaults goes; the keychain stays.'; } },
    { label: 'Install again', when: s => !s.installed, run: s => { s.installed = true; if (s.wipe) { s.keychain = []; return 'First launch: the old keychain items are deleted.'; } return 'The old token is still in the keychain!'; } },
  ],
  switches: [{ key: 'wipe', label: 'Delete leftovers on first launch', value: true }],
  drop: (s, item, from, to) => { s[from] = s[from].filter(x => x !== item); s[to] = s[to].concat([item]); return 'Moved by hand.'; },
  result: s => !s.installed ? { text: 'Not installed', tone: 'neutral' } : s.keychain.length ? { text: 'Signed in', tone: 'good' } : { text: 'Signed out', tone: 'neutral' },
  presets: [{ label: 'Reinstall', actions: ['Launch', 'Delete the app', 'Install again'] }],
});
</script>
